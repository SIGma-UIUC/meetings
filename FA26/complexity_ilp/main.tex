\section{Big O Notation}
\begin{frame}[fragile]{How many computations does this do?}
\begin{pycode}
Algorithm Sum(A, n):
    total = 0
    for i = 1 to n
        total = total + A[i]
    return total
\end{pycode}
\begin{itemize}
\item How many additions does this algorithm perform on an array of $n$ values?
\pause
\item \textbf{Answer:} Exactly $n$ additions. Doubling the input doubles this count.
\end{itemize}
\end{frame}

\begin{frame}[fragile]{What changes with two loops?}
\begin{pycode}
Algorithm ComparePairs(A, n):
    for i = 1 to n
        for j = i + 1 to n
            compare(A[i], A[j])
\end{pycode}
\begin{itemize}
\item How many comparisons does this algorithm perform?
\pause
\item \textbf{Answer:} $(n-1)+(n-2)+\cdots+1=\frac{n(n-1)}{2}$ comparisons.
\pause
\item For large $n$, doubling the input roughly quadruples this count.
\end{itemize}
\end{frame}

\begin{frame}[fragile]{What if the input keeps shrinking?}
\begin{pycode}
Algorithm Halve(n):
    while n > 1
        n = floor(n / 2)
\end{pycode}
\begin{itemize}
\item For an integer $n\geq1$, how many times does the loop divide by two?
\pause
\item \textbf{Answer:} $\lfloor\log_2 n\rfloor$ times. For $n=16$, the values are $16,8,4,2,1$, so there are four divisions.
\pause
\item Let $T(n)$ count the number of operations. Big O describes how that count grows with the input size.
\end{itemize}
\end{frame}

\begin{frame}{Big O notation}
\begin{itemize}
\item We write $T(n)=O(g(n))$ if constants $c>0$ and $n_0$ exist such that
\[0\leq T(n)\leq c\,g(n)\qquad\text{for all }n\geq n_0.\]
\pause
\item The constant $c$ allows a fixed multiplicative difference, and $n_0$ marks where the bound begins to hold.
\pause
\item For example, $3n+7\leq4n$ whenever $n\geq7$. Therefore, $3n+7=O(n)$.
\pause
\item Big O gives an upper bound. A linear runtime is also $O(n^2)$, although that bound is less useful.
\end{itemize}
\end{frame}

\begin{frame}[fragile]{A convex hull recurrence}
\begin{itemize}
\item Consider a divide-and-conquer convex hull algorithm on $n$ points sorted by $x$-coordinate.
\end{itemize}
\pause
\begin{pycode}
Hull(P):
    if |P| <= 3: return the hull of P
    split P into left and right halves
    H_left  = Hull(left)
    H_right = Hull(right)
    return MergeHulls(H_left, H_right)
\end{pycode}
\pause
\begin{itemize}[<+->]
\item The recursive calls solve two subproblems of size $n/2$. The merge takes $O(n)$ time.
\item This gives the recurrence $T(n)=2T(n/2)+O(n)$.
\end{itemize}
\end{frame}

\begin{frame}{The recursion tree}
\centering
\begin{tikzpicture}[>=stealth, level distance=1.15cm,

  edge from parent/.style={draw,->}]
\node {Hull($n$)}
  child {node {Hull($n/2$)}
    child {node {Hull($n/4$)}}
    child {node {Hull($n/4$)}}}
  child {node {Hull($n/2$)}
    child {node {Hull($n/4$)}}
    child {node {Hull($n/4$)}}};
\node[draw=none,fill=none,left=0.7cm of current bounding box.west,align=right,font=\small] {level 0\\[0.82cm]level 1\\[0.82cm]level 2};
\end{tikzpicture}
\begin{itemize}[<+->]
\item At level 0, one merge handles $n$ boundary edges.
\item At level 1, two merges each handle at most $n/2$ edges, for at most $n$ total.
\item Each lower level also performs at most $O(n)$ total merge work. There are $O(\log n)$ levels until subproblems have at most three points.
\item Therefore $T(n)=O(n\log n)$.
\end{itemize}
\end{frame}

\section{Linear Programming}
\frame{\sectionpage}
\begin{frame}{A production problem}
\begin{itemize}
\item A workshop makes tables and chairs. It has 7 planks of wood and enough time to assemble 5 items.
\pause
\end{itemize}
\begin{center}
\begin{tabular}{lcc}
 & Table & Chair\\\hline
Wood required & 2 planks & 1 plank\\
Profit & 3 dollars & 2 dollars
\end{tabular}
\end{center}
\pause
\begin{itemize}
\item Let $x$ count tables and $y$ count chairs. Ignoring whole-number restrictions for now, the LP is
\[
\begin{aligned}
\max\quad &3x+2y\\
\text{subject to}\quad &2x+y\leq7,\qquad x+y\leq5,\\
&x,y\in\mathbb R_{\geq0}.
\end{aligned}
\]
\end{itemize}
\end{frame}

\begin{column}{0.53\textwidth}
\begin{itemize}
\item Each inequality defines a half-plane in two dimensions, or a halfspace in $d$ dimensions. Their intersection is the \textbf{feasible region}.
\pause
\item Here, $x,y\geq0$, $2x+y\leq7$, and $x+y\leq5$ define the shaded region.
\pause
\item To maximize $3x+2y$, move the line $3x+2y=t$ toward larger $t$. The last contact occurs at $(2,3)$, with value $12$.
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Input and output of Seidel's LP algorithm}
\begin{itemize}[<+->]
\item \textbf{Input:} half-plane constraints $H=\{h_1,\ldots,h_n\}$, an objective direction $c$, and a bounding box $B$ containing the feasible region. Each constraint keeps one side of a line (a two-dimensional halfspace).
\item \textbf{Goal:} find an allowed point $(x,y)$ that maximizes a linear score such as $3x+2y$.
\item \textbf{Output:} the best point, or a report that no allowed point exists.
\item Seidel adds the constraints in random order while maintaining the best point for those seen so far.
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Seidel's two-dimensional algorithm}
\begin{smallpycode}
Seidel2D(H, c, B):
    randomly order H as h_1, ..., h_n
    p = corner of B maximizing c dot p
    for i = 1 to n:
        if p violates constraint h_i:
            p = BestOnLine(h_i, H[1..i-1], B, c)
            if p is infeasible: return infeasible
    return p

BestOnLine(h, P, B, c):
    I = boundary(h) intersect B
    for g in P: I = I intersect g
    if I is empty: return infeasible
    return the endpoint of I with the best score
\end{smallpycode}
\pause
\begin{itemize}\small
\item \texttt{BestOnLine} intersects the new boundary with the earlier constraints and returns its best endpoint.
\end{itemize}
\end{frame}

\begin{frame}{Seidel Example: adding one constraint}
\begin{itemize}[<+->]
\item Suppose we know the best point for the first $m-1$ half-planes, then add one more half-plane.
\item If the old optimum satisfies it, nothing changes.
\item Otherwise, the new best point lies on the new boundary line. We scan that line using the earlier constraints.
\end{itemize}
\end{frame}

\begin{frame}{Adding one constraint}
\centering
\includegraphics[width=0.62\textwidth,height=0.76\textheight,keepaspectratio]{images/seidel_lp_insert.png}

\end{frame}

\begin{frame}{Seidel example: a new constraint}
\begin{itemize}[<+->]
\item Maximize $x$ inside a bounding box $0\leq x,y\leq10$, with constraints
\[
C_1:\ y\geq2
\qquad\text{and}\qquad
C_2:\ y\leq5.
\]
\item After $C_1$ and $C_2$, $(10,2)$ is an optimal point.
\item Add
\[
C_3:\ x+y\leq8.
\]
The old point violates it because $10+2>8$.
\item Therefore, the new best point lies somewhere on the boundary line $x+y=8$.
\end{itemize}
\end{frame}

\begin{frame}{Seidel example: the one-dimensional problem}
\begin{itemize}[<+->]
\item On the new boundary, $x=8-y$. Thus maximizing $x$ means making $y$ as small as possible.
\item The earlier constraints give
\[
2\leq y\leq5.
\]
\item This is the feasible part of the new boundary line. Its smallest value is $y=2$, which gives
\[
\boxed{(x,y)=(6,2)}.
\]
\item $C_3$ finds the line by rechecking $C_1$ and $C_2$ (this is an example of a recursive call).
\end{itemize}
\end{frame}

\begin{frame}{Why the two-dimensional algorithm is fast}
\begin{itemize}[<+->]
\item At step $i$, testing whether the current point satisfies the new half-plane takes $O(1)$ time.
\item At most two boundary lines are needed to identify the final best point. In a random order, the $i$th line has probability at most $2/i$ of changing that point.
\item If it changes the point, scanning the $i-1$ earlier constraints along its boundary line takes $O(i)$ time.
\item The expected extra work at step $i$ is therefore at most $\frac{2}{i}\cdot O(i)=O(1)$.
\end{itemize}
\end{frame}

\begin{frame}{Seidel's running time}
\begin{itemize}[<+->]
\item There are $n$ steps, and each costs expected $O(1)$ time. Thus Seidel's two-dimensional algorithm takes expected $O(n)$ time.
\item (More generally, in any fixed dimension $d$, the expected running time is $O(d!\,n)$. The factorial dependence is why this method is mainly used in low dimensions.)
\end{itemize}
\end{frame}

\section{Integer Linear Programming}
\frame{\sectionpage}
\begin{frame}{From LP to integer linear programming}
\begin{itemize}
\item An \textbf{Integer Linear Program (ILP)} adds integer restrictions to an LP. Counts must be whole numbers, and yes-or-no decisions use $0$ or $1$.
\pause
\[
\begin{array}{ll}
\text{LP:} & \min c^Tx\quad\text{subject to }Ax\leq b,\quad x\in\mathbb R^d,\\[0.25cm]
\text{ILP:} & \min c^Tx\quad\text{subject to }Ax\leq b,\quad x\in\mathbb Z^d.
\end{array}
\]
\pause
\item The inequalities still define the same region. The ILP only allows its integer points.
\pause
\item LP has polynomial-time algorithms, but general ILP is NP-hard.
\end{itemize}
\end{frame}
\begin{frame}{An ILP example}
\begin{itemize}[<+->]
\item A workshop makes tables ($x$) and chairs ($y$). It has four units of each of two resources. A table uses $(2,1)$ units and a chair uses $(1,2)$ units.
\item The integer linear program is
\[
\begin{aligned}
\max\quad &3x+2y\\
\text{subject to}\quad &2x+y\leq4,\\
&x+2y\leq4,\\
&x,y\in\mathbb Z_{\geq0}.
\end{aligned}
\]
\item The LP \textit{relaxation} (ignoring integer constraints) has $x=y=\frac43$ and value $\frac{20}{3}$. The ILP optimum is $(x,y)=(2,0)$ with value $6$.
\end{itemize}
\end{frame}

\begin{frame}{How branch and bound works}
\begin{itemize}[<+->]
\item First ignore the integer restrictions and solve the LP relaxation. Its objective value bounds every integer solution in that branch.
\item If the LP solution is fractional, choose one fractional variable and split the search into two cases, such as $x\leq\lfloor x^*\rfloor$ and $x\geq\lceil x^*\rceil$.
\item Whenever the solver finds a valid integer solution, it stores its value as the incumbent.
\item Prune a branch when it is infeasible, already integer, or its LP bound cannot improve the incumbent.
\item Once every branch is solved or pruned, the incumbent is an optimal ILP solution.
\end{itemize}
\end{frame}

\begin{frame}{Branch and bound: search and prune}
\begin{itemize}[<+->]\small
\item The LP relaxation has upper bound $\frac{20}{3}$ and fractional value $x=\frac43$.
\item Branch into $x\leq1$ and $x\geq2$.
\item The $x\geq2$ branch gives $(2,0)$ with value $6$. It becomes the incumbent.
\item The $x\leq1$ branch has LP upper bound $6$, so it cannot improve the incumbent and is pruned.
\end{itemize}

\begin{frame}{ILP solvers in Python}
\begin{itemize}[<+->]
\item In python, we just have to describe variables, linear constraints, and an objective. A solver performs the search.
\item \texttt{scipy.optimize.milp} provides a direct interface to the HiGHS (High Performance Optimization Software) mixed-integer solver.
\item General ILP solvers combine presolve, LP relaxations, cutting planes (another algorithm), heuristics, and branch-and-bound.
\item When the solver reports optimal status, it has proved that no better feasible integer solution exists, subject to numerical tolerances.
\end{itemize}
\end{frame}
