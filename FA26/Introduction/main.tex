\begin{frame}[fragile]{Brute Force}
    \begin{pycode}
def upper_hull_1(points: set[Point]) -> list[Point]:
    lines = []
    for p1, p2 in combinations(points, 2):
        p1, p2 = max(p1, p2), min(p1, p2)
        if all(point_below_line((p1, p2), q)
               for q in points):
            lines.append((p1, p2))
    return sorted(set(flatten(lines)), reverse=True)
    \end{pycode}
\end{frame}

\begin{frame}[fragile]{A little smarter...}

def upper_hull_2(points: set[Point]) -> list[Point]:
    lines = []
    for p1 in points - {min(points)}:
        p2 = max((q for q in points if q < p1),
                  key=lambda q: sin_angle(p1, q))
        if all(point_below_line((p1, p2), q)
               for q in points):
            lines.append((p1, p2))
    return sorted(set(flatten(lines)), reverse=True)
    \end{pycode}

\begin{frame}[fragile]{Sweeping}
    \begin{pycode}
def upper_hull_3(points: set[Point]) -> list[Point]:
    hull = []
    for point in sorted(points, reverse=True):
        while len(hull) >= 2 and
              not point_below_line(
                (hull[-2], hull[-1]), point):
            hull.pop()
        hull.append(point)
    return hull
    \end{pycode}
\end{frame}

\section{CG:SHOP}
\frame{\sectionpage}

\begin{frame}{What is CG:SHOP?}
    \begin{itemize}
        \item \textbf{CG:SHOP} (Computational Geometry: Solving Hard Optimization Problems) is a computation geometry competition for students and researchers.
        \item Each year, teams build solvers for a difficult geometric optimization problem.
        \item The 2027 challenge is \textbf{Multi-Robot Lawn Mowing for Polyominoes}.
    \end{itemize}

\begin{frame}{CG:SHOP Logistics}
    \begin{itemize}
        \item \href{https://cgshop.ibr.cs.tu-bs.de/competition/cg-shop-2027/}{challenge website}.
        \item (Soft) Pre-requisites: data-structures and algorithms at the CS 225 level.
        \item Roughly split into \textit{algorithm} and \textit{implementation} teams.
              \begin{itemize}
                  \item \textit{Algorithm} teams focus on literature review, understanding proofs, and proposing algorithms.
                  \item \textit{Implementation} teams focus on implementing algorithms, and proposing algorithms.
                  \item Both teams need a strong understanding of current research in this area.
              \end{itemize}
    \end{itemize}
